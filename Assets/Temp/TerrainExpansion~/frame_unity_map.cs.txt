var scene = UnityEditor.SceneManagement.EditorSceneManager.GetActiveScene();
if (scene.path != "Assets/Test/CheonMountainMap/Scenes/Test_CheonMountainMap.unity")
    throw new InvalidOperationException("새 테스트 맵 씬이 아닙니다.");
var root = GameObject.Find("Test_CheonMountainMap");
var renderers = root.GetComponentsInChildren<Renderer>();
var bounds = renderers[0].bounds;
foreach (var renderer in renderers.Skip(1)) bounds.Encapsulate(renderer.bounds);
var camera = GameObject.Find("Test_MainCamera").GetComponent<Camera>();
camera.transform.position = bounds.center + new Vector3(-155, 200, 225);
camera.transform.LookAt(bounds.center);
float minX = float.PositiveInfinity, minY = float.PositiveInfinity;
float maxX = float.NegativeInfinity, maxY = float.NegativeInfinity;
foreach (var renderer in renderers)
    for (int x = -1; x <= 1; x += 2)
        for (int y = -1; y <= 1; y += 2)
            for (int z = -1; z <= 1; z += 2)
            {
                var point = camera.transform.InverseTransformPoint(renderer.bounds.center + Vector3.Scale(renderer.bounds.extents, new Vector3(x, y, z)));
                minX = Mathf.Min(minX, point.x); maxX = Mathf.Max(maxX, point.x);
                minY = Mathf.Min(minY, point.y); maxY = Mathf.Max(maxY, point.y);
            }
camera.transform.position += camera.transform.right * ((minX + maxX) * .5f) + camera.transform.up * ((minY + maxY) * .5f);
camera.orthographicSize = Mathf.Max((maxY - minY) * .5f, (maxX - minX) / (2 * (4f / 3f))) * 1.08f;
UnityEditor.SceneManagement.EditorSceneManager.MarkSceneDirty(scene);
UnityEditor.SceneManagement.EditorSceneManager.SaveScene(scene);
var view = UnityEditor.SceneView.lastActiveSceneView;
if (view != null) view.LookAt(bounds.center, camera.transform.rotation, Mathf.Max(bounds.size.x, bounds.size.z) * .6f, true, true);
return new { scene = scene.path, cameraPosition = camera.transform.position.ToString(), cameraSize = camera.orthographicSize };
