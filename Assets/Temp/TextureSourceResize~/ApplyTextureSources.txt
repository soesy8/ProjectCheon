if (UnityEditor.EditorApplication.isPlayingOrWillChangePlaymode)
    throw new InvalidOperationException("편집 모드에서 원본 텍스처를 교체해야 합니다.");

var root = System.IO.Path.GetFullPath(UnityEngine.Application.dataPath + "/..");
var assetsRoot = System.IO.Path.GetFullPath(UnityEngine.Application.dataPath) + System.IO.Path.DirectorySeparatorChar;
var stagingRoot = System.IO.Path.GetFullPath(System.IO.Path.Combine(root, "Assets/Temp/TextureSourceResize~/Staged")) + System.IO.Path.DirectorySeparatorChar;
var plan = Newtonsoft.Json.Linq.JArray.Parse(System.IO.File.ReadAllText("Assets/Temp/TextureSourceResize~/ResizePlan.json"));
Func<string, string> hash = path =>
{
    using (var algorithm = System.Security.Cryptography.SHA256.Create())
    using (var stream = System.IO.File.OpenRead(path))
        return BitConverter.ToString(algorithm.ComputeHash(stream)).Replace("-", "");
};

// 모든 대상과 해시를 먼저 검사하여 준비 중 사용자가 바꾼 파일을 덮어쓰지 않는다.
foreach (var row in plan)
{
    var destination = System.IO.Path.GetFullPath(System.IO.Path.Combine(root, (string)row["path"]));
    var staged = System.IO.Path.GetFullPath((string)row["stagedPath"]);
    if (!destination.StartsWith(assetsRoot, StringComparison.OrdinalIgnoreCase) ||
        !staged.StartsWith(stagingRoot, StringComparison.OrdinalIgnoreCase))
        throw new InvalidOperationException("프로젝트 또는 임시 작업 폴더를 벗어난 경로입니다.");
    if (hash(destination) != (string)row["beforeHash"] || hash(staged) != (string)row["stagedHash"] ||
        hash(destination + ".meta") != (string)row["metaHash"])
        throw new InvalidOperationException("파일이 준비 이후 변경되었습니다: " + (string)row["path"]);
}

UnityEditor.AssetDatabase.StartAssetEditing();
try
{
    foreach (var row in plan)
    {
        string path = (string)row["path"];
        System.IO.File.Copy((string)row["stagedPath"], System.IO.Path.Combine(root, path), true);
        UnityEditor.AssetDatabase.ImportAsset(path, UnityEditor.ImportAssetOptions.ForceUpdate);
    }
}
finally
{
    UnityEditor.AssetDatabase.StopAssetEditing();
}

foreach (var row in plan)
{
    string path = (string)row["path"];
    if (hash(path) != (string)row["stagedHash"] || hash(path + ".meta") != (string)row["metaHash"])
        throw new InvalidOperationException("원본 교체 후 파일 또는 메타 검증에 실패했습니다: " + path);
    var imported = UnityEditor.AssetDatabase.LoadAssetAtPath<UnityEngine.Texture>(path);
    if (imported == null || imported.width > 1024 || imported.height > 1024)
        throw new InvalidOperationException("축소 텍스처가 올바르게 임포트되지 않았습니다: " + path);
}

// 프로젝트 용량 감소가 상쇄되지 않도록 검증을 마친 축소 중간 파일만 제거한다.
foreach (var row in plan)
    System.IO.File.Delete((string)row["stagedPath"]);

return new
{
    resized = plan.Count,
    beforeChangedBytes = plan.Sum(row => (long)row["beforeBytes"]),
    afterChangedBytes = plan.Sum(row => (long)row["bytes"]),
    metaFilesPreserved = true,
    stagedImagesRemoved = true
};
