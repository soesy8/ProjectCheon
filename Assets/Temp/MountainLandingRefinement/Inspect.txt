var scene=UnityEngine.SceneManagement.SceneManager.GetSceneByPath("Assets/Test/CheonMountainMap/Scenes/Test_CheonMountainMap.unity");
if(!scene.isLoaded || EditorApplication.isPlaying)throw new Exception("Mountain scene must be loaded in Edit Mode");
var temp="Assets/Temp/MountainLandingRefinement";
if(!System.IO.File.Exists(temp+"/SceneBefore.unity.txt")){
 var backup=temp+"/SceneBefore.unity";UnityEditor.SceneManagement.EditorSceneManager.SaveScene(scene,backup,true);
 System.IO.File.Move(backup,backup+".txt");if(System.IO.File.Exists(backup+".meta"))System.IO.File.Move(backup+".meta",backup+".txt.meta");
}
var map=scene.GetRootGameObjects().Single(r=>r.name=="Test_CheonMountainMap");
Func<Vector3,float[]> V=v=>new[]{v.x,v.y,v.z};
var stairReport=new List<object>();
foreach(var name in new[]{"Stairway_Between_Summits","Stairway_Winding_Approach"}){
 var t=map.transform.Find(name);var mesh=t.GetComponent<MeshFilter>().sharedMesh;var vs=mesh.vertices;var uv=mesh.uv;var indices=mesh.triangles;
 var faces=new List<object>();
 for(int i=0;i<indices.Length;i+=3){var ids=new[]{indices[i],indices[i+1],indices[i+2]};var a=t.TransformPoint(vs[ids[0]]);var b=t.TransformPoint(vs[ids[1]]);var c=t.TransformPoint(vs[ids[2]]);
  if(Vector3.Cross(b-a,c-a).normalized.y>.98f){var center=(a+b+c)/3;faces.Add(new {y=center.y,position=V(center),vertices=ids.Select(id=>V(t.TransformPoint(vs[id]))).ToArray(),uv=ids.Select(id=>new[]{uv[id].x,uv[id].y}).ToArray()});}
 }
 stairReport.Add(new{name,material=AssetDatabase.GetAssetPath(t.GetComponent<MeshRenderer>().sharedMaterial),topFaces=faces.OrderByDescending(f=>(float)f.GetType().GetProperty("y").GetValue(f)).Take(20).ToArray()});
}
var colliders=map.GetComponentsInChildren<Collider>();var probes=new List<object>();
for(int z=-20;z<=0;z+=2)for(int x=-32;x<=-12;x+=2){var hits=Physics.RaycastAll(new Vector3(x,55,z),Vector3.down,35).Where(h=>h.collider.gameObject.scene==scene).OrderByDescending(h=>h.point.y).Select(h=>new{name=h.collider.name,y=h.point.y}).ToArray();probes.Add(new {x,z,hits});}
var report=new {dirty=scene.isDirty,stairs=stairReport,landings=map.transform.Find("StairStoneLandings").Cast<Transform>().Select(t=>new{name=t.name,pos=V(t.position),scale=V(t.localScale)}).ToArray(),walls=map.transform.Find("GateSideWalls").GetComponentsInChildren<Transform>().Where(t=>t.name.EndsWith("_End")).Select(t=>new{name=t.name,position=V(t.position)}).ToArray(),probes};
var json=Newtonsoft.Json.JsonConvert.SerializeObject(report,Newtonsoft.Json.Formatting.Indented);System.IO.File.WriteAllText(temp+"/Inspection.json",json);return "Saved current scene and stair UV/terrain inspection";
