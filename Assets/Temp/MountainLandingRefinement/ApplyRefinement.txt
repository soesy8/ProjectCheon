  float x0=-.5f+col/6f,x1=-.5f+(col+1)/6f,z0=row/8f,z1=(row+1)/8f,b=.002f;
  var a=P(x0+b,0,z0+b);var bb=P(x0+b,0,z1-b);var c=P(x1-b,0,z1-b);var d=P(x1-b,0,z0+b);
  var aa=P(x0,-.004f,z0);var bc=P(x0,-.004f,z1);var cc=P(x1,-.004f,z1);var dd=P(x1,-.004f,z0);
  var uv=patches[(row*17+col*31)%patches.Count];
  Quad(a,bb,c,d,uv);Quad(aa,bc,bb,a,uv);Quad(bb,bc,cc,c,uv);Quad(d,c,cc,dd,uv);Quad(aa,a,d,dd,uv);
  var ab=P(x0,-1,z0);var cb=P(x0,-1,z1);var cd=P(x1,-1,z1);var da=P(x1,-1,z0);
  Quad(ab,cb,bc,aa,uv);Quad(bc,cb,cd,cc,uv);Quad(dd,cc,cd,da,uv);Quad(ab,aa,dd,da,uv);Quad(da,cd,cb,ab,uv);
 }
 var result=new Mesh();result.SetVertices(vertices);result.SetUVs(0,uvs);result.SetTriangles(triangles,0);result.RecalculateNormals();result.RecalculateBounds();return result;
