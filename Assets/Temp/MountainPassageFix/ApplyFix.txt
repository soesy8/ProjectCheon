Action<Vector3,Vector3,Vector3,Vector3,Vector2,Vector2> Quad=(a,b,c,d,uvMin,uvMax)=>{
 int n=vertices.Count;vertices.AddRange(new[]{a,b,c,d});
 uv.AddRange(new[]{uvMin,new Vector2(uvMin.x,uvMax.y),uvMax,new Vector2(uvMax.x,uvMin.y)});

};

for(int row=0;row<6;row++)for(int column=0;column<4;column++) {
 float x0=-.5f+column/4f,x1=-.5f+(column+1)/4f,z0=row/6f,z1=(row+1)/6f;
 float bevel=.0035f,lower=-.007f;
 var a=new Vector3(x0+bevel,0,z0+bevel);var b=new Vector3(x0+bevel,0,z1-bevel);

 var aa=new Vector3(x0,lower,z0);var bb=new Vector3(x0,lower,z1);var cc=new Vector3(x1,lower,z1);var dd=new Vector3(x1,lower,z0);
 var min=new Vector2((column*.173f+row*.097f)%.5f,(row*.117f+column*.071f)%.5f);var max=min+new Vector2(.45f,.45f);
 Quad(a,b,c,d,min,max);
 Quad(aa,bb,b,a,min,max);Quad(b,bb,cc,c,min,max);Quad(d,c,cc,dd,min,max);Quad(aa,a,d,dd,min,max);
 var ab=new Vector3(x0,-1,z0);var bc=new Vector3(x0,-1,z1);var cd=new Vector3(x1,-1,z1);var da=new Vector3(x1,-1,z0);
 Quad(ab,bc,bb,aa,min,max);Quad(bb,bc,cd,cc,min,max);Quad(dd,cc,cd,da,min,max);Quad(ab,aa,dd,da,min,max);
 Quad(da,cd,bc,ab,min,max);
}
