var scene = UnityEngine.SceneManagement.SceneManager.GetSceneByPath("Assets/Test/CheonMountainMap/Scenes/Test_CheonMountainMap.unity");
var cameraObject = new GameObject("MountainPlacementPreviewCamera");
cameraObject.hideFlags = HideFlags.HideAndDontSave;
UnityEngine.SceneManagement.SceneManager.MoveGameObjectToScene(cameraObject, scene);
var camera = cameraObject.AddComponent<Camera>();
var texture = new RenderTexture(1024, 600, 24, RenderTextureFormat.ARGB32);
texture.antiAliasing = 4;
var previous = RenderTexture.active;
try
{
    camera.orthographic = true;
    camera.orthographicSize = 72f;
    camera.nearClipPlane = 0.1f;
    camera.farClipPlane = 1000f;
    camera.clearFlags = CameraClearFlags.SolidColor;
    camera.backgroundColor = new Color(0.48f, 0.60f, 0.63f);
    camera.allowHDR = true;
    camera.allowMSAA = true;
    camera.transform.rotation = Quaternion.Euler(40f, 132f, 0f);
    var target = new Vector3(-12f, 30f, -38f);
    camera.transform.position = target - camera.transform.forward * 300f;
    var data = cameraObject.AddComponent<UnityEngine.Rendering.Universal.UniversalAdditionalCameraData>();
    data.renderPostProcessing = false;
    data.renderShadows = true;
    camera.targetTexture = texture;
    camera.Render();
    RenderTexture.active = texture;
    var image = new Texture2D(1024, 600, TextureFormat.RGB24, false);
    image.ReadPixels(new Rect(0, 0, 1024, 600), 0, 0);
    image.Apply();
    System.IO.File.WriteAllBytes("Assets/Temp/MountainPassageFix/FinalOverview.png", image.EncodeToPNG());
    UnityEngine.Object.DestroyImmediate(image);
    return "Saved 1024 x 600 overview";
}
finally
{
    camera.targetTexture = null;
    RenderTexture.active = previous;
    texture.Release();
    UnityEngine.Object.DestroyImmediate(texture);
    UnityEngine.Object.DestroyImmediate(cameraObject);
}
