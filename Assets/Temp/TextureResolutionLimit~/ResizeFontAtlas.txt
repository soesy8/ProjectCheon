const string fontPath = "Assets/Test/CheonMountainMap/Arts/Fonts/Test_MountainPlayerKorean.asset";
const int maximumSize = 1024;
var font = UnityEditor.AssetDatabase.LoadAssetAtPath<TMPro.TMP_FontAsset>(fontPath);
var characters = font.characterTable.Select(character => character.unicode).Distinct().ToArray();
var serializedFont = new UnityEditor.SerializedObject(font);
var sourceGuid = serializedFont.FindProperty("m_SourceFontFileGUID").stringValue;
var source = UnityEditor.AssetDatabase.LoadAssetAtPath<UnityEngine.Font>(UnityEditor.AssetDatabase.GUIDToAssetPath(sourceGuid));
if (source == null)
    throw new InvalidOperationException("아틀라스를 다시 생성할 원본 폰트가 없습니다.");

var atlas = font.atlasTextures[0];
var atlasIdBefore = UnityEditor.GlobalObjectId.GetGlobalObjectIdSlow(atlas).ToString();
var materialIdBefore = UnityEditor.GlobalObjectId.GetGlobalObjectIdSlow(font.material).ToString();
var originalMode = font.atlasPopulationMode;
var originalMultiAtlas = font.isMultiAtlasTexturesEnabled;
var features = UnityEngine.JsonUtility.ToJson(font.fontFeatureTable);
System.IO.File.WriteAllText("Assets/Temp/TextureResolutionLimit~/FontCharacters.before.json",
    Newtonsoft.Json.JsonConvert.SerializeObject(new { fontPath, characters, atlasIdBefore, materialIdBefore }));

// 원본을 수정하기 전에 임시 폰트로 기존 글자가 1024 아틀라스 한 장에 모두 들어가는지 확인한다.
var candidate = TMPro.TMP_FontAsset.CreateFontAsset(source, (int)font.faceInfo.pointSize,
    font.atlasPadding, font.atlasRenderMode, maximumSize, maximumSize, TMPro.AtlasPopulationMode.Dynamic, false);
try
{
    candidate.TryAddCharacters(characters, out uint[] candidateMissing);
    var unavailable = characters.Where(character => !candidate.HasCharacter((int)character)).ToArray();
    if (unavailable.Length > 0)
        throw new InvalidOperationException("1024 아틀라스에 기존 문자를 모두 넣을 수 없습니다: " + string.Join(",", unavailable));

    // 기존 폰트, 머티리얼, 기본 아틀라스 객체를 재사용해 씬의 GUID와 fileID 참조를 보존한다.
    typeof(TMPro.TMP_FontAsset).GetProperty("SourceFont_EditorRef",
        System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic).SetValue(font, source);
    font.atlasPopulationMode = TMPro.AtlasPopulationMode.Dynamic;
    serializedFont.Update();
    serializedFont.FindProperty("m_AtlasWidth").intValue = maximumSize;
    serializedFont.FindProperty("m_AtlasHeight").intValue = maximumSize;
    serializedFont.ApplyModifiedPropertiesWithoutUndo();
    font.ClearFontAssetData();
    font.TryAddCharacters(characters, out uint[] missing);
    font.atlasPopulationMode = originalMode;
    font.isMultiAtlasTexturesEnabled = originalMultiAtlas;
    UnityEngine.JsonUtility.FromJsonOverwrite(features, font.fontFeatureTable);

    var remaining = characters.Where(character => !font.HasCharacter((int)character)).ToArray();
    if (remaining.Length > 0)
        throw new InvalidOperationException("재생성 후 누락된 문자가 있습니다: " + string.Join(",", remaining));

    var materials = UnityEditor.AssetDatabase.FindAssets("t:Material", new[] { "Assets" })
        .Select(UnityEditor.AssetDatabase.GUIDToAssetPath).Distinct()
        .SelectMany(path => UnityEditor.AssetDatabase.LoadAllAssetsAtPath(path).OfType<UnityEngine.Material>())
        .Where(material => material.HasProperty("_MainTex") && material.GetTexture("_MainTex") == atlas)
        .Append(font.material).Distinct().ToArray();
    foreach (var material in materials)
    {
        if (material.HasProperty("_TextureWidth")) material.SetFloat("_TextureWidth", maximumSize);
        if (material.HasProperty("_TextureHeight")) material.SetFloat("_TextureHeight", maximumSize);
        UnityEditor.EditorUtility.SetDirty(material);
        UnityEditor.AssetDatabase.SaveAssetIfDirty(material);
    }

    UnityEditor.EditorUtility.SetDirty(font);
    UnityEditor.EditorUtility.SetDirty(atlas);
    UnityEditor.AssetDatabase.SaveAssetIfDirty(font);
    TMPro.TMPro_EventManager.ON_FONT_PROPERTY_CHANGED(true, font);

    var atlasIdAfter = UnityEditor.GlobalObjectId.GetGlobalObjectIdSlow(atlas).ToString();
    var materialIdAfter = UnityEditor.GlobalObjectId.GetGlobalObjectIdSlow(font.material).ToString();
    return new
    {
        fontPath,
        width = atlas.width,
        height = atlas.height,
        previousCharacterCount = characters.Length,
        characterCount = font.characterTable.Count,
        missing = remaining,
        atlasReferencePreserved = atlasIdBefore == atlasIdAfter,
        materialReferencePreserved = materialIdBefore == materialIdAfter
    };
}
finally
{
    foreach (var temporaryAtlas in candidate.atlasTextures)
        if (temporaryAtlas != null) UnityEngine.Object.DestroyImmediate(temporaryAtlas);
    if (candidate.material != null) UnityEngine.Object.DestroyImmediate(candidate.material);
    UnityEngine.Object.DestroyImmediate(candidate);
}
