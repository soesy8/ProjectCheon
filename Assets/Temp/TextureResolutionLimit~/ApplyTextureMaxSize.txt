if (UnityEditor.EditorApplication.isPlayingOrWillChangePlaymode)
    throw new InvalidOperationException("편집 모드에서 텍스처 설정을 변경해야 합니다.");

const int maximumSize = 1024;
var paths = UnityEditor.AssetDatabase.FindAssets("t:Texture", new[] { "Assets" })
    .Select(UnityEditor.AssetDatabase.GUIDToAssetPath)
    .Distinct()
    .Where(path => UnityEditor.AssetImporter.GetAtPath(path) is UnityEditor.TextureImporter)
    .OrderBy(path => path)
    .ToArray();
var changedPaths = new List<string>();

// 임포트 작업을 모아서 처리하고, 예외가 발생해도 에디터의 에셋 편집 잠금을 해제한다.
UnityEditor.AssetDatabase.StartAssetEditing();
try
{
    foreach (var path in paths)
    {
        var importer = (UnityEditor.TextureImporter)UnityEditor.AssetImporter.GetAtPath(path);
        var metaText = System.IO.File.ReadAllText(path + ".meta");
        var targets = System.Text.RegularExpressions.Regex.Matches(metaText, @"(?m)^\s+buildTarget:\s*([^\r\n]+)")
            .Cast<System.Text.RegularExpressions.Match>()
            .Select(match => match.Groups[1].Value.Trim())
            .Append("DefaultTexturePlatform")
            .Distinct()
            .ToArray();

        importer.maxTextureSize = maximumSize;
        foreach (var target in targets)
        {
            var settings = target == "DefaultTexturePlatform"
                ? importer.GetDefaultPlatformTextureSettings()
                : importer.GetPlatformTextureSettings(target);
            // 압축 방식, 플랫폼 Override 여부 등 기존 옵션은 보존하고 크기만 변경한다.
            settings.maxTextureSize = maximumSize;
            importer.SetPlatformTextureSettings(settings);
        }

        importer.SaveAndReimport();
        changedPaths.Add(path);
    }
}
finally
{
    UnityEditor.AssetDatabase.StopAssetEditing();
}

var violations = paths.Select(path => new
{
    path,
    texture = UnityEditor.AssetDatabase.LoadAssetAtPath<UnityEngine.Texture>(path),
    importer = (UnityEditor.TextureImporter)UnityEditor.AssetImporter.GetAtPath(path)
}).Where(item => item.texture == null || item.texture.width > maximumSize || item.texture.height > maximumSize ||
    item.importer.GetDefaultPlatformTextureSettings().maxTextureSize != maximumSize)
    .Select(item => item.path).ToArray();

return new { changed = changedPaths.Count, maximumSize, violations };
