var paths = UnityEditor.AssetDatabase.FindAssets("t:Mesh", new [] { "Assets" }).Concat(UnityEditor.AssetDatabase.FindAssets("t:Model", new [] { "Assets" })).Select(UnityEditor.AssetDatabase.GUIDToAssetPath).Where(p => !p.StartsWith("Assets/Temp/")).Distinct().OrderBy(p => p).ToArray();
var meshes = new System.Collections.Generic.List<object>();
var models = new System.Collections.Generic.List<object>();
string ObjectKey(UnityEngine.Object obj) { if (obj == null) return "null"; UnityEditor.AssetDatabase.TryGetGUIDAndLocalFileIdentifier(obj, out string guid, out long id); return guid+":"+id; }
string Digest(byte[] data) { using(var hash=System.Security.Cryptography.SHA256.Create()) return System.BitConverter.ToString(hash.ComputeHash(data)).Replace("-", "").ToLowerInvariant(); }
foreach(var path in paths) {
    foreach(var mesh in UnityEditor.AssetDatabase.LoadAllAssetsAtPath(path).OfType<UnityEngine.Mesh>()) {
        using(var dataArray = UnityEditor.MeshUtility.AcquireReadOnlyMeshData(mesh)) {
            var data = dataArray[0];
            long bytes = 0;
            using(var stream = new System.IO.MemoryStream()) {
                for(int i=0; i<mesh.vertexBufferCount; i++) { var buffer = data.GetVertexData<byte>(i).ToArray(); bytes += buffer.Length; stream.Write(buffer,0,buffer.Length); }
                var indices = data.GetIndexData<byte>().ToArray(); bytes += indices.Length; stream.Write(indices,0,indices.Length);
                var info = new System.Text.StringBuilder();
                foreach(var attr in mesh.GetVertexAttributes()) info.Append(attr.ToString()).Append(';');
                for(int i=0; i<mesh.subMeshCount; i++) { var sub=data.GetSubMesh(i); info.AppendFormat("{0},{1},{2},{3},{4},{5};", sub.indexStart,sub.indexCount,sub.baseVertex,sub.firstVertex,sub.vertexCount,(int)sub.topology); }
                var extra = System.Text.Encoding.UTF8.GetBytes(info.ToString()); stream.Write(extra,0,extra.Length);
                meshes.Add(new { path, key=ObjectKey(mesh), name=mesh.name, vertices=mesh.vertexCount, subMeshes=mesh.subMeshCount, blendShapes=mesh.blendShapeCount, readable=mesh.isReadable, bufferBytes=bytes, memoryBytes=UnityEngine.Profiling.Profiler.GetRuntimeMemorySizeLong(mesh), geometryHash=Digest(stream.ToArray()) });
            }
        }
    }
    var importer=UnityEditor.AssetImporter.GetAtPath(path) as UnityEditor.ModelImporter;
    if(importer != null) {
        var shape = new System.Text.StringBuilder();
        var root=UnityEditor.AssetDatabase.LoadAssetAtPath<UnityEngine.GameObject>(path);
        if(root != null) foreach(var tr in root.GetComponentsInChildren<UnityEngine.Transform>(true)) {
            shape.Append(ObjectKey(tr)).Append('|').Append(tr.name).Append('|').Append(ObjectKey(tr.parent)).Append('|').Append(tr.localPosition.ToString("R")).Append('|').Append(tr.localRotation.ToString("R")).Append('|').Append(tr.localScale.ToString("R")).Append(';');
            var filter=tr.GetComponent<UnityEngine.MeshFilter>(); if(filter != null) shape.Append(ObjectKey(filter.sharedMesh));
            var renderer=tr.GetComponent<UnityEngine.Renderer>(); if(renderer != null) foreach(var mat in renderer.sharedMaterials) shape.Append('|').Append(ObjectKey(mat));
            var skin=tr.GetComponent<UnityEngine.SkinnedMeshRenderer>(); if(skin != null) { shape.Append(ObjectKey(skin.sharedMesh)); foreach(var bone in skin.bones) shape.Append('|').Append(ObjectKey(bone)); }
        }
        models.Add(new { path, readable=importer.isReadable, shapeHash=Digest(System.Text.Encoding.UTF8.GetBytes(shape.ToString())), materials=importer.GetExternalObjectMap().OrderBy(x=>x.Key.name).Select(x=>x.Key.name+"="+ObjectKey(x.Value)).ToArray() });
    }
}
var loadedMeshes = UnityEngine.Resources.FindObjectsOfTypeAll<UnityEngine.Mesh>();
return new { playing=UnityEditor.EditorApplication.isPlaying, scene=UnityEngine.SceneManagement.SceneManager.GetActiveScene().path, dirty=UnityEngine.SceneManagement.SceneManager.GetActiveScene().isDirty, meshes, models, loadedMeshCount=loadedMeshes.Length, loadedMeshBytes=loadedMeshes.Sum(m=>UnityEngine.Profiling.Profiler.GetRuntimeMemorySizeLong(m)) };
