var meshes=UnityEngine.Object.FindObjectsByType<UnityEngine.MeshFilter>(UnityEngine.FindObjectsInactive.Include,UnityEngine.FindObjectsSortMode.None).Select(x=>x.sharedMesh).Concat(UnityEngine.Object.FindObjectsByType<UnityEngine.SkinnedMeshRenderer>(UnityEngine.FindObjectsInactive.Include,UnityEngine.FindObjectsSortMode.None).Select(x=>x.sharedMesh)).Concat(UnityEngine.Object.FindObjectsByType<UnityEngine.MeshCollider>(UnityEngine.FindObjectsInactive.Include,UnityEngine.FindObjectsSortMode.None).Select(x=>x.sharedMesh)).Where(x=>x!=null).Distinct().ToArray();
var entries=meshes.Select(m=>{ UnityEditor.AssetDatabase.TryGetGUIDAndLocalFileIdentifier(m,out string guid,out long id); return new { name=m.name, key=guid+":"+id, path=UnityEditor.AssetDatabase.GetAssetPath(m), readable=m.isReadable, bytes=UnityEngine.Profiling.Profiler.GetRuntimeMemorySizeLong(m) }; }).OrderBy(x=>x.key).ToArray();
return new { playing=UnityEditor.EditorApplication.isPlaying, frame=UnityEngine.Time.frameCount, time=UnityEngine.Time.time, meshes=entries, meshCount=entries.Length, meshBytes=entries.Sum(x=>x.bytes), readableCount=entries.Count(x=>x.readable) };
