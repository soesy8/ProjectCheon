if(UnityEditor.EditorApplication.isPlaying) throw new System.InvalidOperationException("편집 모드에서만 실행합니다.");
var protectedPaths=new System.Collections.Generic.HashSet<string>(new [] { "Assets/Arts/Environment/CheonNoiseMap/GraniteBoulder.asset", "Assets/Arts/Environment/SelfModeling/Models/Wall_End.fbx", "Assets/Arts/Environment/SelfModeling/Models/Wall_Middle.fbx", "Assets/Test/CheonMountainMap/Models/Test_CheonMountainMap.fbx" });
UnityEditor.AssetDatabase.Refresh(UnityEditor.ImportAssetOptions.ForceSynchronousImport);
var nativePaths=System.IO.Directory.GetFiles("Assets/Temp/ModelMemoryOptimization~/ReadWriteBackup/Assets", "*.asset",System.IO.SearchOption.AllDirectories).Select(p=>p.Replace('\\','/').Substring("Assets/Temp/ModelMemoryOptimization~/ReadWriteBackup/".Length)).OrderBy(p=>p).ToArray();
var changedNative=new System.Collections.Generic.List<string>();
foreach(var path in nativePaths) {
    if(protectedPaths.Contains(path)) throw new System.InvalidOperationException("보호된 충돌 메시: "+path);
    var mesh=UnityEditor.AssetDatabase.LoadAssetAtPath<UnityEngine.Mesh>(path);
    if(mesh == null) throw new System.InvalidOperationException("메시 로드 실패: "+path);
    var serialized=new UnityEditor.SerializedObject(mesh);
    var readable=serialized.FindProperty("m_IsReadable");
    if(readable == null) throw new System.InvalidOperationException("메시 설정 누락: "+path);
    if(readable.boolValue) {
        readable.boolValue=false;
        serialized.ApplyModifiedPropertiesWithoutUndo();
        UnityEditor.EditorUtility.SetDirty(mesh);
        UnityEditor.AssetDatabase.SaveAssetIfDirty(mesh);
        UnityEditor.AssetDatabase.ImportAsset(path,UnityEditor.ImportAssetOptions.ForceUpdate);
    }
    if(UnityEditor.AssetDatabase.LoadAssetAtPath<UnityEngine.Mesh>(path).isReadable) throw new System.InvalidOperationException("설정 저장 실패: "+path);
    changedNative.Add(path);
}
var changedModels=new System.Collections.Generic.List<string>();
foreach(var guid in UnityEditor.AssetDatabase.FindAssets("t:Model",new [] { "Assets" })) {
    var path=UnityEditor.AssetDatabase.GUIDToAssetPath(guid);
    if(path.StartsWith("Assets/Temp/") || protectedPaths.Contains(path)) continue;
    var importer=UnityEditor.AssetImporter.GetAtPath(path) as UnityEditor.ModelImporter;
    if(importer != null && importer.isReadable) { importer.isReadable=false; importer.SaveAndReimport(); changedModels.Add(path); }
}
return new { nativeCount=changedNative.Count, modelCount=changedModels.Count, changedNative, changedModels };
