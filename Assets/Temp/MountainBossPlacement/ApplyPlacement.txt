var scenePath = "Assets/Test/CheonMountainMap/Scenes/Test_CheonMountainMap.unity";
var scene = UnityEngine.SceneManagement.SceneManager.GetSceneByPath(scenePath);
if (!scene.IsValid() || !scene.isLoaded || EditorApplication.isPlaying)
    throw new Exception("The mountain scene must be loaded in Edit Mode.");
var map = scene.GetRootGameObjects().Single(root => root.name == "Test_CheonMountainMap");
var player = scene.GetRootGameObjects().Single(root => root.name == "Test_MountainPlayer");
var cameraObject = scene.GetRootGameObjects().Single(root => root.name == "Test_MainCamera");
var gateMesh = map.transform.Find("Hanok_Gate_Placement").GetComponentInChildren<MeshFilter>();
var arenaFloor = map.transform.Find("Arena_Floor_Flat_UV").GetComponent<MeshCollider>();
var shrineFloor = map.transform.Find("Shrine_Floor_Flat_UV").GetComponent<MeshCollider>();
var mountainCore = map.transform.Find("Mountain_Continuous_Core").GetComponent<MeshCollider>();
var wallAsset = AssetDatabase.LoadAssetAtPath<GameObject>("Assets/Prefabs/Environment/SelfModeling/Wall_Middle.prefab");
var capAsset = AssetDatabase.LoadAssetAtPath<GameObject>("Assets/Prefabs/Environment/SelfModeling/Wall_End.prefab");
var bossAsset = AssetDatabase.LoadAssetAtPath<GameObject>("Assets/Prefabs/Bosses/JangseungBoss.prefab");
if (wallAsset == null || capAsset == null || bossAsset == null)
    throw new Exception("Required placement prefab is missing.");
if (map.transform.Find("GateSideWalls") != null || scene.GetRootGameObjects().Any(root => root.name == "JangseungBoss"))
    throw new Exception("Encounter placement already exists; inspect it before applying again.");

var tempPath = "Assets/Temp/MountainBossPlacement";
if (!System.IO.File.Exists(tempPath + "/SceneBefore.unity.txt"))
{
    var backupPath = tempPath + "/SceneBefore.unity";
    UnityEditor.SceneManagement.EditorSceneManager.SaveScene(scene, backupPath, true);
    System.IO.File.Move(backupPath, tempPath + "/SceneBefore.unity.txt");
    if (System.IO.File.Exists(backupPath + ".meta"))
        System.IO.File.Move(backupPath + ".meta", tempPath + "/SceneBefore.unity.txt.meta");
}
Func<Vector3, float[]> V = value => new[] { value.x, value.y, value.z };
Func<MeshCollider, Vector3, Vector3> OnFloor = (floor, position) =>
{
    RaycastHit hit;
    var ray = new Ray(new Vector3(position.x, 200f, position.z), Vector3.down);
    if (!floor.Raycast(ray, out hit, 400f) && !mountainCore.Raycast(ray, out hit, 400f))
        throw new Exception("Placement is outside terrain: " + position);
    return hit.point;
};
var previousPlayer = player.transform.position;
var playerPosition = OnFloor(shrineFloor, new Vector3(15.75f, 0f, 25.10f)) + Vector3.up * 0.04f;
var bossPosition = OnFloor(arenaFloor, new Vector3(-28.25f, 0f, -67.90f));
var gateCenter = gateMesh.transform.TransformPoint(new Vector3(0f, 0.0005f, 0f));
gateCenter.y = 37.12f;
var gateAxis = Vector3.ProjectOnPlane(gateMesh.transform.right, Vector3.up).normalized;
var upperStart = gateCenter + gateAxis * 4.45f;
var lowerStart = gateCenter - gateAxis * 4.45f;
var upperPath = new[] {
    upperStart, new Vector3(-1.5f, 0f, -22.72f), new Vector3(2f, 0f, -20.5f),
    new Vector3(6f, 0f, -17f), new Vector3(10f, 0f, -14f)
};
var lowerPath = new[] {
    lowerStart, new Vector3(-10.8f, 0f, -13.6f),
    new Vector3(-12.8f, 0f, -11.8f), new Vector3(-14.8f, 0f, -10f)
};
for (int i = 0; i < upperPath.Length; i++) upperPath[i] = OnFloor(arenaFloor, upperPath[i]);
for (int i = 0; i < lowerPath.Length; i++) lowerPath[i] = OnFloor(arenaFloor, lowerPath[i]);

Undo.IncrementCurrentGroup();
int undoGroup = Undo.GetCurrentGroup();
Undo.SetCurrentGroupName("산 맵 플레이어·장승·문 옆 담 배치");
var wallReport = new List<object>();
GameObject boss = null;
GameObject walls = null;
try
{
    Undo.RecordObject(player.transform, "신당 앞 플레이어 배치");
    var controller = player.GetComponent<CharacterController>();
    bool controllerEnabled = controller != null && controller.enabled;
    if (controllerEnabled) controller.enabled = false;
    var facingGate = Vector3.ProjectOnPlane(gateCenter - playerPosition, Vector3.up).normalized;
    player.transform.SetPositionAndRotation(playerPosition, Quaternion.LookRotation(facingGate));
    if (controllerEnabled) controller.enabled = true;
    PrefabUtility.RecordPrefabInstancePropertyModifications(player.transform);

    boss = (GameObject)PrefabUtility.InstantiatePrefab(bossAsset, scene);
    Undo.RegisterCreatedObjectUndo(boss, "광장 중앙 장승 보스 배치");
    boss.name = "JangseungBoss";
    boss.transform.SetPositionAndRotation(bossPosition,
        Quaternion.LookRotation(Vector3.ProjectOnPlane(gateCenter - bossPosition, Vector3.up).normalized));
    boss.transform.localScale = Vector3.one;
    var bossCollider = Undo.AddComponent<CapsuleCollider>(boss);
    bossCollider.height = 5.43f;
    bossCollider.radius = 0.66f;
    bossCollider.center = new Vector3(0f, 2.715f, 0f);
    PrefabUtility.RecordPrefabInstancePropertyModifications(boss.transform);

    walls = new GameObject("GateSideWalls");
    Undo.RegisterCreatedObjectUndo(walls, "문 옆 담 배치");
    Undo.SetTransformParent(walls.transform, map.transform, "담 그룹 정리");
    foreach (var run in new[] { new { name = "Upper", points = upperPath }, new { name = "Lower", points = lowerPath } })
    {
        var group = new GameObject("GateWall_" + run.name);
        Undo.RegisterCreatedObjectUndo(group, "담 구간");
        Undo.SetTransformParent(group.transform, walls.transform, "담 구간 정리");
        int number = 0;
        for (int edge = 0; edge < run.points.Length - 1; edge++)
        {
            var a = run.points[edge];
            var b = run.points[edge + 1];
            int count = Mathf.CeilToInt(Vector3.Distance(a, b) / 2.4f);
            for (int segment = 0; segment < count; segment++)
            {
                var start = OnFloor(arenaFloor, Vector3.Lerp(a, b, segment / (float)count));
                var end = OnFloor(arenaFloor, Vector3.Lerp(a, b, (segment + 1f) / count));
                var direction = (end - start).normalized;
                var wallForward = Vector3.Cross(-direction, Vector3.up).normalized;
                var wallUp = Vector3.Cross(wallForward, -direction).normalized;
                var item = (GameObject)PrefabUtility.InstantiatePrefab(wallAsset, scene);
                Undo.RegisterCreatedObjectUndo(item, "담 중간 부재");
                item.name = "GateWall_" + run.name + "_" + (++number).ToString("00");
                item.transform.SetParent(group.transform, false);
                // 원본 담의 길이는 로컬 -X 방향 2.4m이므로 시작점에서 진행 방향으로 맞춘다.
                item.transform.SetPositionAndRotation(start - Vector3.up * 0.025f,
                    Quaternion.LookRotation(wallForward, wallUp));
                item.transform.localScale = new Vector3((Vector3.Distance(start, end) + 0.07f) / 2.4f, 1.8f, 1f);
                foreach (var filter in item.GetComponentsInChildren<MeshFilter>())
                {
                    if (filter.GetComponent<Collider>() != null) continue;
                    var collider = Undo.AddComponent<MeshCollider>(filter.gameObject);
                    collider.sharedMesh = filter.sharedMesh;
                }
                PrefabUtility.RecordPrefabInstancePropertyModifications(item.transform);
                wallReport.Add(new { name = item.name, start = V(start), end = V(end) });
            }
        }
        var tip = run.points[run.points.Length - 1];
        var tangent = (tip - run.points[run.points.Length - 2]).normalized;
        var cap = (GameObject)PrefabUtility.InstantiatePrefab(capAsset, scene);
        Undo.RegisterCreatedObjectUndo(cap, "담 끝 부재");
        cap.name = "GateWall_" + run.name + "_End";
        cap.transform.SetParent(group.transform, false);
        cap.transform.SetPositionAndRotation(tip - Vector3.up * 0.025f,
            Quaternion.LookRotation(Vector3.Cross(tangent, Vector3.up), Vector3.up));
        cap.transform.localScale = new Vector3(1f, 1.8f, 1f);
        foreach (var filter in cap.GetComponentsInChildren<MeshFilter>())
        {
            if (filter.GetComponent<Collider>() != null) continue;
            var collider = Undo.AddComponent<MeshCollider>(filter.gameObject);
            collider.sharedMesh = filter.sharedMesh;
        }
        PrefabUtility.RecordPrefabInstancePropertyModifications(cap.transform);
    }

    Undo.RecordObject(cameraObject.transform, "플레이어 시작 위치 카메라 정렬");
    var cameraControl = cameraObject.GetComponent<Cheon.Testing.MountainMap.Test_MountainPlayerCamera>();
    if (cameraControl != null) cameraControl.Configure(player.transform);
    PrefabUtility.RecordPrefabInstancePropertyModifications(cameraObject.transform);
    Physics.SyncTransforms();
    UnityEditor.SceneManagement.EditorSceneManager.MarkSceneDirty(scene);
    if (!UnityEditor.SceneManagement.EditorSceneManager.SaveScene(scene))
        throw new Exception("Mountain scene save failed.");
    Undo.CollapseUndoOperations(undoGroup);

    var report = new {
        scene = scene.path,
        previousPlayer = V(previousPlayer),
        player = V(player.transform.position),
        playerForward = V(player.transform.forward),
        boss = V(boss.transform.position),
        bossForward = V(boss.transform.forward),
        gateCenter = V(gateCenter),
        wallSections = wallReport.Count,
        wallEndCaps = 2,
        upperPath = upperPath.Select(V).ToArray(),
        lowerPath = lowerPath.Select(V).ToArray(),
        walls = wallReport
    };
    var json = Newtonsoft.Json.JsonConvert.SerializeObject(report, Newtonsoft.Json.Formatting.Indented);
    System.IO.File.WriteAllText(tempPath + "/PlacementReport.json", json);
    return json;
}
catch
{
    Undo.RevertAllDownToGroup(undoGroup);
    throw;
}
