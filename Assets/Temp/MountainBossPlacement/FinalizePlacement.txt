if (EditorApplication.isPlaying) throw new Exception("Return to Edit Mode first.");
var settings = Newtonsoft.Json.Linq.JObject.Parse(System.IO.File.ReadAllText("Assets/Temp/MountainBossPlacement/PlaySettingsBefore.json"));
Application.runInBackground = (bool)settings["runInBackground"];
var tickType = AppDomain.CurrentDomain.GetAssemblies()
    .Select(assembly => assembly.GetType("Unity.Pipeline.Editor.Commands.AutoTickCommand")).First(type => type != null);
tickType.GetMethod("SetAutoTick", System.Reflection.BindingFlags.Static | System.Reflection.BindingFlags.Public)
    .Invoke(null, new object[] { (bool)settings["autoTickEnabled"], (int)settings["autoTickInterval"], false });

var scene = UnityEngine.SceneManagement.SceneManager.GetSceneByPath("Assets/Test/CheonMountainMap/Scenes/Test_CheonMountainMap.unity");
var all = scene.GetRootGameObjects().SelectMany(root => root.GetComponentsInChildren<Transform>(true)).ToArray();
Func<Transform, string> PathOf = null;
PathOf = transform => transform.parent == null ? transform.name : PathOf(transform.parent) + "/" + transform.name;
var byPath = all.ToDictionary(PathOf);
var before = Newtonsoft.Json.Linq.JObject.Parse(System.IO.File.ReadAllText("Assets/Temp/MountainBossPlacement/SceneBefore.json"));
var unexpected = new List<string>();
foreach (var item in before["matches"])
{
    string path = (string)item["path"];
    if (path.StartsWith("Test_MountainPlayer") || path == "Test_MainCamera") continue;
    Transform current;
    if (!byPath.TryGetValue(path, out current)) { unexpected.Add("Missing: " + path); continue; }
    var coordinates = item["position"].Select(value => (float)value).ToArray();
    var expected = new Vector3(coordinates[0], coordinates[1], coordinates[2]);
    if (Vector3.Distance(expected, current.position) > 0.002f)
        unexpected.Add("Moved: " + path);
}
if (unexpected.Count != 0) throw new Exception(string.Join("; ", unexpected));
var player = scene.GetRootGameObjects().Single(root => root.name == "Test_MountainPlayer");
var boss = scene.GetRootGameObjects().Single(root => root.name == "JangseungBoss");
var walls = byPath["Test_CheonMountainMap/GateSideWalls"].gameObject;
Func<Vector3, float[]> V = value => new[] { value.x, value.y, value.z };
var report = new {
    scene = scene.path, saved = !scene.isDirty,
    player = V(player.transform.position), boss = V(boss.transform.position),
    originalObjectsMovedOrMissing = unexpected.Count,
    originalShrineWallRenderers = byPath["Test_ShrineEntranceWalls"].GetComponentsInChildren<Renderer>().Length,
    newGateWallRenderers = walls.GetComponentsInChildren<Renderer>().Length,
    newGateWallColliders = walls.GetComponentsInChildren<Collider>().Length
};
System.IO.File.WriteAllText("Assets/Temp/MountainBossPlacement/FinalVerification.json",
    Newtonsoft.Json.JsonConvert.SerializeObject(report, Newtonsoft.Json.Formatting.Indented));

AssetDatabase.Refresh(ImportAssetOptions.ForceSynchronousImport);
foreach (var path in AssetDatabase.FindAssets("t:Texture2D", new[] { "Assets/Temp/MountainBossPlacement" })
    .Select(AssetDatabase.GUIDToAssetPath))
{
    var importer = AssetImporter.GetAtPath(path) as TextureImporter;
    if (importer == null) continue;
    importer.maxTextureSize = 1024;
    foreach (var platform in new[] { "DefaultTexturePlatform", "Standalone", "Android", "iPhone", "WebGL" })
    {
        var platformSettings = importer.GetPlatformTextureSettings(platform);
        platformSettings.maxTextureSize = 1024;
        importer.SetPlatformTextureSettings(platformSettings);
    }
    importer.SaveAndReimport();
}
Selection.objects = new UnityEngine.Object[] { player, boss, walls };
var sceneView = SceneView.lastActiveSceneView;
if (sceneView != null)
{
    sceneView.sceneLighting = true;
    sceneView.LookAt(new Vector3(-12f, 30f, -38f), Quaternion.Euler(40f, 132f, 0f), 80f, true, false);
    sceneView.Repaint();
}
return Newtonsoft.Json.JsonConvert.SerializeObject(report, Newtonsoft.Json.Formatting.Indented);
