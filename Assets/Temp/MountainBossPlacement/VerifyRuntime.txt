if (!EditorApplication.isPlaying) throw new Exception("Play mode is required.");
var player = GameObject.Find("Test_MountainPlayer");
var boss = GameObject.Find("JangseungBoss");
var walls = GameObject.Find("Test_CheonMountainMap/GateSideWalls");
var map = GameObject.Find("Test_CheonMountainMap");
if (player == null || boss == null || walls == null) throw new Exception("A placed object is missing.");
Func<Vector3, float[]> V = value => new[] { value.x, value.y, value.z };
var placement = Newtonsoft.Json.Linq.JObject.Parse(System.IO.File.ReadAllText("Assets/Temp/MountainBossPlacement/PlacementReport.json"));
var expectedPlayer = placement["player"].Select(token => (float)token).ToArray();
var expectedBoss = placement["boss"].Select(token => (float)token).ToArray();
var playerExpected = new Vector3(expectedPlayer[0], expectedPlayer[1], expectedPlayer[2]);
var bossExpected = new Vector3(expectedBoss[0], expectedBoss[1], expectedBoss[2]);
var controller = player.GetComponent<CharacterController>();
var movement = player.GetComponent<Cheon.Testing.MountainMap.Test_MountainPlayerMovement>();
var spawn = (Vector3)movement.GetType().GetField("spawnPosition",
    System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic).GetValue(movement);
var shrineFloor = map.transform.Find("Shrine_Floor_Flat_UV").GetComponent<MeshCollider>();
RaycastHit ground;
if (!shrineFloor.Raycast(new Ray(player.transform.position + Vector3.up * 10f, Vector3.down), out ground, 30f))
    throw new Exception("Player is no longer over the shrine courtyard.");
float horizontalDrift = Vector3.ProjectOnPlane(player.transform.position - playerExpected, Vector3.up).magnitude;
if (horizontalDrift > 0.1f || !controller.isGrounded)
    throw new Exception("Player did not settle at the requested start.");
if (Vector3.Distance(boss.transform.position, bossExpected) > 0.01f)
    throw new Exception("Boss moved from its saved placement.");
var gateCenterValues = placement["gateCenter"].Select(token => (float)token).ToArray();
var gateCenter = new Vector3(gateCenterValues[0], gateCenterValues[1], gateCenterValues[2]);
var normal = new Vector3(0.7016157f, 0f, 0.7125556f);
var passageHits = Physics.SphereCastAll(gateCenter + normal * 4f + Vector3.up, 0.35f, -normal, 8f,
    ~0, QueryTriggerInteraction.Ignore).Where(hit => hit.collider.transform.IsChildOf(walls.transform)).ToArray();
if (passageHits.Length != 0) throw new Exception("New walls obstruct the gate passage.");
if (walls.GetComponentsInChildren<Collider>().Length != 16)
    throw new Exception("A wall collider is missing.");
var report = new {
    frame = Time.frameCount, player = V(player.transform.position), spawn = V(spawn),
    grounded = controller.isGrounded, horizontalDrift,
    playerFootToGround = controller.bounds.min.y - ground.point.y,
    boss = V(boss.transform.position), bossHeight = 5.43f,
    wallColliders = walls.GetComponentsInChildren<Collider>().Length,
    passageBlockedByNewWalls = passageHits.Length > 0
};
var json = Newtonsoft.Json.JsonConvert.SerializeObject(report, Newtonsoft.Json.Formatting.Indented);
System.IO.File.WriteAllText("Assets/Temp/MountainBossPlacement/RuntimeVerification.json", json);
return json;
