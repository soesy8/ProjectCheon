   var end=landing.position+landing.right*lane+landing.forward*((float)layout["landings"].Single(p=>(string)p["name"]==landing.name)["length"]+1.4f);

    if(body.isGrounded&&vertical<0)vertical=-2;vertical-=24*dt;

    var delta=player.transform.position-before;

    if(!body.isGrounded)airborne++;minimumY=Mathf.Min(minimumY,player.transform.position.y);biggestFall=Mathf.Max(biggestFall,-delta.y);
   }
   var remaining=Vector3.ProjectOnPlane(end-player.transform.position,Vector3.up).magnitude;
   reports.Add(new {landing=landing.name,lane,reverse,reached=remaining<.2f,remaining,blockedFrames=blocked,airborneFrames=airborne,minimumY,biggestFall,start=V(start),end=V(player.transform.position)});
  }
 }
}finally{body.enabled=false;player.transform.SetPositionAndRotation(savedPosition,savedRotation);body.enabled=true;movement.enabled=movementEnabled;Physics.SyncTransforms();}
