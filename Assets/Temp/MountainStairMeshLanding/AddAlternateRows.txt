 var parents=new int[vertices.Length];var welded=new Dictionary<string,int>();
 for(int i=0;i<vertices.Length;i++){var p=t.TransformPoint(vertices[i]);var key=Math.Round(p.x,4)+","+Math.Round(p.y,4)+","+Math.Round(p.z,4);int previous;parents[i]=welded.TryGetValue(key,out previous)?previous:i;welded[key]=parents[i];}

 float highest=components.Max(g=>g.top);float selectedHeight=components.Where(g=>g.top<highest-.1f).Max(g=>g.top);

 float height=selectedHeight-selected.Min(i=>t.TransformPoint(vertices[i]).y);var origin=center+rotation*new Vector3((minX+maxX)*.5f,0,maxZ);

  int offset=outputVertices.Count;

  for(int i=0;i<indices.Length;i+=3)if(selectedRoots.Contains(Find(indices[i]))){outputIndices.Add(offset+lookup[indices[i]]);outputIndices.Add(offset+lookup[indices[i+1]]);outputIndices.Add(offset+lookup[indices[i+2]]);}
 }
 var path=((string)area["mesh"]).Replace(".asset","Solid.asset");var mesh=new Mesh{name=System.IO.Path.GetFileNameWithoutExtension(path)};mesh.SetVertices(outputVertices);mesh.SetNormals(outputNormals);mesh.SetUVs(0,outputUvs);mesh.SetTriangles(outputIndices,0);mesh.RecalculateBounds();AssetDatabase.CreateAsset(mesh,path);
 var group=map.transform.Find("StairLandingExtensions/"+(string)area["name"]);
 for(int row=1;row<group.childCount;row+=2){var item=group.GetChild(row);var filter=item.GetComponent<MeshFilter>();var collider=item.GetComponent<MeshCollider>();Undo.RecordObject(filter,"통돌 계단 석재 번갈아 배치");Undo.RecordObject(collider,"통돌 발판 충돌");filter.sharedMesh=mesh;collider.sharedMesh=mesh;PrefabUtility.RecordPrefabInstancePropertyModifications(filter);PrefabUtility.RecordPrefabInstancePropertyModifications(collider);}
 area["alternateMesh"]=path;
}
