var scene=UnityEngine.SceneManagement.SceneManager.GetSceneByPath("Assets/Test/CheonMountainMap/Scenes/Test_CheonMountainMap.unity");
if(EditorApplication.isPlaying||!scene.isLoaded)throw new Exception("Edit Mode required");
var temp="Assets/Temp/MountainStairMeshLanding";
if(!System.IO.File.Exists(temp+"/SceneBefore.unity.txt")){
 var backup=temp+"/SceneBefore.unity";UnityEditor.SceneManagement.EditorSceneManager.SaveScene(scene,backup,true);
 System.IO.File.Move(backup,backup+".txt");if(System.IO.File.Exists(backup+".meta"))System.IO.File.Move(backup+".meta",backup+".txt.meta");
}
var map=scene.GetRootGameObjects().Single(r=>r.name=="Test_CheonMountainMap");var report=new List<object>();
Func<Vector3,float[]> V=v=>new[]{v.x,v.y,v.z};
foreach(var name in new[]{"Stairway_Winding_Approach","Stairway_Between_Summits"}){
 var t=map.transform.Find(name);var mesh=t.GetComponent<MeshFilter>().sharedMesh;var vertices=mesh.vertices;var triangles=mesh.triangles;
 var welded=new Dictionary<string,int>();var parents=new int[vertices.Length];
 for(int i=0;i<vertices.Length;i++){var p=t.TransformPoint(vertices[i]);string key=Math.Round(p.x,4)+","+Math.Round(p.y,4)+","+Math.Round(p.z,4);int previous;parents[i]=welded.TryGetValue(key,out previous)?previous:i;welded[key]=parents[i];}
 Func<int,int> Find=null;Find=i=>parents[i]==i?i:(parents[i]=Find(parents[i]));
 for(int i=0;i<triangles.Length;i+=3){var a=Find(triangles[i]);parents[Find(triangles[i+1])]=a;parents[Find(triangles[i+2])]=a;}
 var groups=Enumerable.Range(0,vertices.Length).GroupBy(Find);float maxY=vertices.Select(v=>t.TransformPoint(v).y).Max();
 foreach(var group in groups){var bounds=new Bounds(t.TransformPoint(vertices[group.First()]),Vector3.zero);foreach(var id in group)bounds.Encapsulate(t.TransformPoint(vertices[id]));
  if(bounds.max.y<maxY-1.05f)continue;
  report.Add(new{name,component=group.Key,vertices=group.Count(),min=V(bounds.min),max=V(bounds.max),center=V(bounds.center),size=V(bounds.size)});
 }
}
var json=Newtonsoft.Json.JsonConvert.SerializeObject(report,Newtonsoft.Json.Formatting.Indented);System.IO.File.WriteAllText(temp+"/StepComponents.json",json);return json;
